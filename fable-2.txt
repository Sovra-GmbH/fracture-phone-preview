ender(renderer);
    const t = this.rtA; this.rtA = this.rtB; this.rtB = t;
    this.compU.tScene.value = this.mrt.textures[0];
    this.compU.tTrail.value = this.rtA.texture;

    this.comp.render(renderer);
  }
  dispose() { this.mrt.dispose(); this.rtA.dispose(); this.rtB.dispose(); }
}
