/* ambient dust: the volume both fields share */
const DUST = (() => {
  const N = 1600;
  const p = new Float32Array(N * 3), r = new Float32Array(N);
  for (let i = 0; i < N; i++) {
    const a = rnd() * Math.PI * 2, e = Math.asin(rnd() * 2 - 1), rr = 1.2 + Math.pow(rnd(), 0.6) * 9.0;
    p.set([Math.cos(a) * Math.cos(e) * rr, Math.sin(e) * rr * 0.8, Math.sin(a) * Math.cos(e) * rr], i * 3);
    r[i] = rnd();
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.BufferAttribute(p, 3));
  g.setAttribute('prand', new THREE.BufferAttribute(r, 1));
  g.boundingSphere = new THREE.Sphere(new THREE.Vector3(), 60);
  const uniforms = { uTime: { value: 0 }, uFade: { value: 1 }, uPx: { value: 1 }, uCool: { value: 0 } };
  const m = new THREE.ShaderMaterial({
    uniforms, transparent: true, blending: THREE.AdditiveBlending, depthWrite: false,
    vertexShader: /* glsl */`
      attribute float prand; uniform float uTime, uPx; varying float vR, vD;
      void main(){
        vec3 p = position;
        p += 0.25 * vec3(sin(uTime*0.11 + prand*40.0), cos(uTime*0.09 + prand*17.0), sin(uTime*0.13 + prand*9.0));
        vec4 mv = modelViewMatrix * vec4(p, 1.0);
        float d = -mv.z;
        gl_PointSize = (1.4 + 2.2*prand) * uPx * clamp(4.0 / max(d, 0.5), 0.35, 3.0);
        gl_Position = projectionMatrix * mv; vR = prand; vD = d;
      }`,
    fragmentShader: /* glsl */`
      uniform float uFade, uCool; varying float vR, vD;
      ${MRT_OUT}
      void main(){
        vec2 q = gl_PointCoord - 0.5; float r2 = dot(q, q); if (r2 > 0.25) discard;
        float a = exp(-r2*14.0);
        vec3 warm = vec3(0.55, 0.5, 0.5), cool = vec3(0.45, 0.6, 0.85);
        vec3 c = mix(warm, cool, uCool) * (0.12 + 0.2*vR);
        float fog = exp(-vD*0.06);
        gl_FragColor = vec4(c * a * fog * uFade * smoothstep(0.2, 0.9, vD), 1.0);
        gSwarm = vec4(0.0); gAll = vec4(0.0);
      }`
  });
  const pts = new THREE.Points(g, m); pts.frustumCulled = false; pts.renderOrder = 1;
  scene.add(pts);
  return { uniforms, pts };
})();
