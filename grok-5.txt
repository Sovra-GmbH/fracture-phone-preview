

const persistMat = pass(`
  uniform sampler2D tCurr, tPrev;
  uniform float uDamp;
  varying vec2 vUv;
  void main(){
    vec3 a = texture2D(tCurr, vUv).rgb;
    vec3 b = texture2D(tPrev, vUv).rgb;
    vec3 c = a + b * uDamp;
    gl_FragColor = vec4(c / (1.0 + c * 0.55), 1.0);
  }
`, { tCurr: { value: null }, tPrev: { value: null }, uDamp: { value: 0.86 } });

const brightMat = pass(`
  uniform sampler2D tMap;
  uniform float uTh;
  varying vec2 vUv;
  void main(){
    vec3 c = texture2D(tMap, vUv).rgb;
    float l = dot(c, vec3(0.2126, 0.7152, 0.0722));
    float b = smoothstep(uTh, uTh + 0.35, l);
    gl_FragColor = vec4(c * b, 1.0);
  }
`, { tMap: { value: null }, uTh: { value: 0.28 } });

const blurMat = pass(`
  uniform sampler2D tMap;
  uniform vec2 uDir;
  varying vec2 vUv;
  void main(){
    vec3 s = texture2D(tMap, vUv).rgb * 0.227027;
    s += texture2D(tMap, vUv + uDir).rgb * 0.1945946;
    s += texture2D(tMap, vUv - uDir).rgb * 0.1945946;
    s += texture2D(tMap, vUv + uDir * 2.0).rgb * 0.1216216;
    s += texture2D(tMap, vUv - uDir * 2.0).rgb * 0.1216216;
    s += texture2D(tMap, vUv + uDir * 3.0).rgb * 0.054054;
    s += texture2D(tMap, vUv - uDir * 3.0).rgb * 0.054054;
    gl_FragColor = vec4(s, 1.0);
  }
`, { tMap: { value: null }, uDir: { value: new THREE.Vector2() } });

const finalMat = pass(`
  uniform sampler2D tScene, tBloom;
  uniform float uBloom, uGrain, uTime, uVig, uFade;
  varying vec2 vUv;
  vec3 aces(vec3 x){
    const float a = 2.51, b = 0.03, c = 2.43, d = 0.59, e = 0.14;
    return clamp((x * (a * x + b)) / (x * (c * x + d) + e), 0.0, 1.0);
  }
  float hash(vec2 p){ return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453); }
  void main(){
    vec2 uv = vUv;
    vec2 c = uv - 0.5;
    float ca = 0.0009 + length(c) * 0.0024;
    vec3 col;
    col.r = texture2D(tScene, uv + vec2(ca, 0.0)).r;
    col.g = texture2D(tScene, uv).g;
    col.b = texture2D(tScene, uv - vec2(ca, 0.0)).b;
    vec3 bl = texture2D(tBloom, uv).rgb;
    col += bl * uBloom;
    col *= uFade;
    col = aces(col * 0.92);
    float vig = 1.0 - dot(c, c) * uVig;
    col *= vig;
    float g = (hash(uv * vec2(1920.0, 1080.0) + uTime * 60.0) - 0.5) * uGrain;
    col += g;
    gl_FragColor = vec4(pow(max(col, 0.0), vec3(0.95)), 1.0);
  }
`, {
  tScene: { value: null },
  tBloom: { value: null },
  uBloom: { value: 0.9 },
  uGrain: { value: 0.035 },
  uTime: { value: 0 },
  uVig: { value: 1.15 },
  uFade: { value: 1 }
});

}

function layoutCaps(t) {
  let best = null, op = 0;
  for (const c of CAPS) {
    const fi = c.voice === "far" ? 0.55 : c.voice === "aftermath" ? 0.85 : 0.2;
    const fo = c.voice === "far" ? 0.7 : c.voice === "aftermath" ? 1.1 : 0.38;
    const o = pulse(t, c.t, c.d, fi, fo);
    if (o > op) { op = o; best = c; }
  }
  if (phaseOf(t) === "portrait" || phaseOf(t) === "recede" || phaseOf(t) === "void" || phaseOf(t) === "crack") {
    capEl.style.opacity = "0";
    return;
  }
  if (!best || op <= 0.01) {
    capEl.style.opacity = "0";
    return;
  }
  if (capEl.textContent !== best.text) capEl.textContent = best.text;
  capEl.className = best.voice;
  capEl.style.opacity = String(op);
}
