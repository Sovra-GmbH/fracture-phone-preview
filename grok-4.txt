const look = new THREE.Vector3();
const camPos = new THREE.Vector3();

function phaseOf(t) {
  if (t < CRACK_AT) return "duel";
  if (t < CRACK_AT + CRACK_DUR) return "crack";
  if (t < CRACK_AT + CRACK_DUR + VOID_DUR) return "void";
  if (t < CRACK_AT + CRACK_DUR + VOID_DUR + PORT_DUR) return "portrait";
  if (t < CRACK_AT + CRACK_DUR + VOID_DUR + PORT_DUR + RECEDE_DUR) return "recede";
  return t < END_AT + 2 ? "after" : "hold";
}

function cameraFor(t) {
  const ph = phaseOf(t);
  if (ph === "void") {
    return;
  }
  if (ph === "portrait" || ph === "recede") {
    const u0 = CRACK_AT + CRACK_DUR + VOID_DUR;
    const u = (t - u0) / (PORT_DUR + RECEDE_DUR);
    const rec = ph === "recede" ? (t - (u0 + PORT_DUR)) / RECEDE_DUR : 0;
    const hand = 0.028 * (1 - rec);
    camPos.set(
      -0.15 + Math.sin(t * 1.3) * hand + rec * 0.9,
      -0.12 + Math.cos(t * 1.05) * hand * 0.8 - rec * 0.05,
      2.15 - Math.min(u, 0.72) * 0.42 + rec * 2.4
    );
    look.set(1.15, 0.04, 0.0);
    camera.fov = 38 + rec * 16;
    camera.updateProjectionMatrix();
    camera.position.copy(camPos);
    camera.lookAt(look);
    camera.rotation.z = (Math.sin(t * 0.7) * 0.03) * (1 - rec);
    return;
  }
  if (ph === "after" || ph === "hold") {
    const k = env(t, LAST_AT, END_AT + 1.5);
    camPos.set(0.4 + k * 1.1, 0.15, 3.4 + k * 1.6);
    look.set(0.2, 0, 0);
    camera.fov = 48;
    camera.updateProjectionMatrix();
    camera.position.copy(camPos);
    camera.lookAt(look);
    return;
  }
  const ne = nearE(t);
  const fe = farE(t);
  const r = 3.15 - ne * 0.7 + fe * 0.32;
  const th = t * (0.105 + ne * 0.27) + 0.4;
  const ph2 = 0.22 + Math.sin(t * 0.09) * 0.28 + fe * 0.08;
  camPos.set(
    Math.cos(th) * r * Math.cos(ph2),
    Math.sin(ph2) * 1.15 + Math.sin(t * 0.17) * 0.18,
    Math.sin(th) * r * Math.cos(ph2)
  );
  look.set(
    Math.sin(t * 0.13) * 0.22,
    Math.cos(t * 0.11) * 0.12,
    Math.sin(t * 0.09) * 0.18
  );
  camera.fov = 50 - ne * 8 + fe * 7;
  camera.updateProjectionMatrix();
  camera.position.copy(camPos);
  camera.lookAt(look);
  camera.rotation.z = ne * 0.05 - fe * 0.02;
  frozen.pos.copy(camera.position);
  frozen.target.copy(look);
  frozen.fov = camera.fov;
  frozen.set = true;
}
