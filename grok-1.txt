function env(t, a, b) {
  if (t <= a) return 0;
  if (t >= b) return 1;
  const x = (t - a) / (b - a);
  return x * x * (3 - 2 * x);
}
function pulse(t, start, dur, fadeIn, fadeOut) {
  if (t < start || t > start + dur) return 0;
  const l = t - start;
  if (l < fadeIn) return l / fadeIn;
  if (l > dur - fadeOut) return Math.max(0, (dur - l) / fadeOut);
  return 1;
}

function fillBall(n, r) {
  const pos = new Float32Array(n * 3);
  const seed = new Float32Array(n);
  for (let i = 0; i < n; i++) {
    let x, y, z, m;
    do {
      x = Math.random() * 2 - 1;
      y = Math.random() * 2 - 1;
      z = Math.random() * 2 - 1;
      m = x * x + y * y + z * z;
    } while (m > 1 || m < 0.0002);
    const k = r * Math.cbrt(Math.random());
    pos[i * 3] = x * k;
    pos[i * 3 + 1] = y * k;
    pos[i * 3 + 2] = z * k;
    seed[i] = Math.random();
  }
  return { pos, seed };
}
function fillShells(n) {
  const pos = new Float32Array(n * 3);
  const seed = new Float32Array(n);
  const golden = Math.PI * (3 - Math.sqrt(5));
  const shells = [1.08, 1.66, 2.24];
  for (let i = 0; i < n; i++) {
    const sh = shells[i % 3];
    const k = Math.floor(i / 3);
    const nn = Math.ceil(n / 3);
    const y = 1 - (k / Math.max(nn - 1, 1)) * 2;
    const rad = Math.sqrt(Math.max(0, 1 - y * y));
    const th = golden * k;
    const jitter = 0.04 + (i % 7) * 0.002;
    pos[i * 3] = Math.cos(th) * rad * sh + (Math.random() - 0.5) * jitter;
    pos[i * 3 + 1] = y * sh + (Math.random() - 0.5) * jitter;
    pos[i * 3 + 2] = Math.sin(th) * rad * sh + (Math.random() - 0.5) * jitter;
    seed[i] = Math.random();
  }
  return { pos, seed };
}
