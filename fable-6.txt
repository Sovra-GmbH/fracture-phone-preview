  const mp = FAR.mainPrism;
  const Y = new THREE.Vector3(0, 1, 0), X = new THREE.Vector3(1, 0, 0);
  const state = { drive: 0, shock: 0, fade: 1 };
  function step(dt, t) {
    const D = state.drive;
    const targetSpeed = 0.9 + 1.7 * D;
    for (let b = 0; b < 3; b++) {
      // slow precession of each stream's axis
      axNow[b].copy(AX[b]).applyAxisAngle(Y, t * (0.05 + 0.02 * b)).applyAxisAngle(X, Math.sin(t * 0.07 + b) * 0.25).normalize();
    }
    const tq = 1.1 + 0.5 * D, turbAmp = 0.35 + 1.5 * D;
    const plane = 6.0 / (1.0 + 2.2 * D);   // the streams hold at low drive and come apart as it accelerates
    const ph1 = t * 0.31, ph2 = t * 0.23, ph3 = t * 0.27;
    const tight = 1.06 - 0.13 * Math.min(D, 2.6);
    const mdx = mp.dir.x, mdy = mp.dir.y, mdz = mp.dir.z, mh0 = mp.base, mh1 = mp.base + mp.len + mp.tip, mr = mp.rad + 0.18;
    const shock = state.shock; state.shock = 0;
    for (let i = 0; i < NP; i++) {
      const o = i * 3;
      let x = pos[o], y = pos[o + 1], z = pos[o + 2];
      let vx = vel[o], vy = vel[o + 1], vz = vel[o + 2];
      const r = Math.sqrt(x * x + y * y + z * z) + 1e-6;
      const nx = x / r, ny = y / r, nz = z / r;
      const A = axNow[band[i]], ax = A.x, ay = A.y, az = A.z;
      // orbit (tangential to the stream's axis) and a spring back into the stream's plane
      const tx = ay * z - az * y, ty = az * x - ax * z, tz = ax * y - ay * x;
      const hh = x * ax + y * ay + z * az;
      // shell spring
      const sr = shellR[i] * tight;
      const rad = -(r - sr) * 7.0;
      // ABC turbulence, two octaves
      const px = x * tq, py = y * tq, pz = z * tq;
      let ux = Math.sin(pz + ph1) + 0.6 * Math.cos(py - ph2);
      let uy = 0.8 * Math.sin(px + ph3) + Math.cos(pz + ph1 * 0.5);
      let uz = 0.6 * Math.sin(py - ph2) + 0.8 * Math.cos(px + ph3 * 0.7);
      const qx = x * 3.9 + 3.1, qy = y * 3.9 - 1.7, qz = z * 3.9 + 0.4;
      ux += 0.22 * (Math.sin(qz - ph3 * 2.0) + 0.6 * Math.cos(qy + ph1));
      uy += 0.22 * (0.8 * Math.sin(qx + ph2 * 1.7) + Math.cos(qz - ph1));
      uz += 0.22 * (0.6 * Math.sin(qy + ph3) + 0.8 * Math.cos(qx - ph2));
      let acx = tx * 1.2 + nx * rad + ux * turbAmp - ax * hh * plane;
      let acy = ty * 1.2 + ny * rad + uy * turbAmp - ay * hh * plane;
      let acz = tz * 1.2 + nz * rad + uz * turbAmp - az * hh * plane;
      // stay out of the main prism (capsule) and the core
      const h = x * mdx + y * mdy + z * mdz;
      const hc = clamp(h, mh0, mh1);
      const cx = x - mdx * hc, cy = y - mdy * hc, cz = z - mdz * hc;
      const cd = Math.sqrt(cx * cx + cy * cy + cz * cz) + 1e-6;
      if (cd < mr) { const k = (mr - cd) / cd * 40.0; acx += cx * k; acy += cy * k; acz += cz * k; }
      if (r < 1.15) { const k = (1.15 - r) * 30.0; acx += nx * k; acy += ny * k; acz += nz * k; }
      vx += acx * dt; vy += acy * dt; vz += acz * dt;
      if (shock > 0) { vx += nx * shock * (0.6 + prand[i]); vy += ny * shock * (0.6 + prand[i]); vz += nz * shock * (0.6 + prand[i]); }
      const sp = Math.sqrt(vx * vx + vy * vy + vz * vz) + 1e-6;
      const want = targetSpeed * spd[i];
      const kk = 1 + (want / sp - 1) * Math.min(1, dt * 12.0);
      vx *= kk; vy *= kk; vz *= kk;
      x += vx * dt; y += vy * dt; z += vz * dt;
      pos[o] = x; pos[o + 1] = y; pos[o + 2] = z; vel[o] = vx; vel[o + 1] = vy; vel[o + 2] = vz;
    }
    head = (head + 1) % NT;
    hist.set(pos, head * NP * 3);
    hPosAttr.needsUpdate = true;
  }
